PLANNED boot script (~/.termux/boot/shizuku-watchdog-boot.sh):

  #!/data/data/com.termux/files/usr/bin/sh
  # REBOOT_PERSISTENCE boot glue — writes its own evidence BEFORE any
  # interactive session, so H2 does not depend on the operator opening Termux.
  LOG=$HOME/ladb-shizuku-recovery/reboot_boot_evidence.log
  {
    echo "=== boot $(date '+%F %T') ==="
    if pgrep -x runsvdir >/dev/null 2>&1; then
      echo "runsvdir already up (not started by boot script)"
    else
      nohup /data/data/com.termux/files/usr/bin/runsvdir \
        /data/data/com.termux/files/usr/var/service >/dev/null 2>&1 &
      echo "runsvdir started by boot script"
    fi
    sleep 3
    /data/data/com.termux/files/usr/bin/sv status \
      /data/data/com.termux/files/usr/var/service/shizuku-watchdog
  } >> "$LOG" 2>&1

  reboot_boot_evidence.log stays untracked until the post-experiment commit
  (same pattern as the force-stop prestate dossier).
